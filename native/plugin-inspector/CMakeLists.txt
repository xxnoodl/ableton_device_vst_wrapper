cmake_minimum_required(VERSION 3.20)
project(studio_inspector LANGUAGES CXX)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
 message(FATAL_ERROR "The initial inspector supports Windows x64 only")
endif()
option(STUDIO_BUILD_TEST_FIXTURES "Build original SDK-ABI test modules (no audio DSP)" ON)
if(STUDIO_BUILD_TEST_FIXTURES)
 foreach(mode IN ITEMS normal crash hang multi)
  add_library(inspector-fixture-${mode} MODULE test-fixture.cpp vendor/pluginterfaces/base/funknown.cpp)
  target_include_directories(inspector-fixture-${mode} PRIVATE vendor)
  target_compile_features(inspector-fixture-${mode} PRIVATE cxx_std_17)
  target_compile_definitions(inspector-fixture-${mode} PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS FIXTURE_${mode})
  target_link_libraries(inspector-fixture-${mode} PRIVATE ole32)
  set_target_properties(inspector-fixture-${mode} PROPERTIES PREFIX "" SUFFIX ".vst3")
 endforeach()
endif()
add_executable(studio-inspector inspector.cpp vendor/pluginterfaces/base/funknown.cpp)
target_include_directories(studio-inspector PRIVATE vendor)
target_compile_features(studio-inspector PRIVATE cxx_std_17)
target_compile_definitions(studio-inspector PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN _CRT_SECURE_NO_WARNINGS)
target_link_libraries(studio-inspector PRIVATE ole32 user32)
if(MSVC)
 target_compile_options(studio-inspector PRIVATE /W4 /EHsc /utf-8)
endif()
